──────────────────────────────────────────────────────
  Accessing the services
──────────────────────────────────────────────────────
{{- if .Values.ingress.enabled }}

  Access via Ingress:
  {{- range .Values.ingress.hosts }}
  http{{ if $.Values.ingress.tls }}s{{ end }}://{{ .host }}
  {{- end }}
{{- else if .Values.ingressRoute.enabled }}

  Access via Traefik IngressRoute (check your route match rules).
{{- else }}

  Ingress is disabled. Use port-forward to access locally:

  # Backend API
  kubectl port-forward -n {{ .Release.Namespace }} \
    svc/{{ include "teslasync.fullname" . }} {{ .Values.service.port }}:{{ .Values.service.port }}

  {{- if .Values.web.enabled }}
  # Web frontend
  kubectl port-forward -n {{ .Release.Namespace }} \
    svc/{{ include "teslasync.fullname" . }}-web {{ .Values.web.service.port }}:{{ .Values.web.service.port }}
  {{- end }}

  {{- if .Values.grafana.enabled }}
  # Grafana
  kubectl port-forward -n {{ .Release.Namespace }} \
    svc/{{ include "teslasync.fullname" . }}-grafana {{ .Values.grafana.service.port }}:{{ .Values.grafana.service.port }}
  {{- end }}

  Then open:
    Backend API : http://localhost:{{ .Values.service.port }}/healthz
    {{- if .Values.web.enabled }}
    Web UI      : http://localhost:{{ .Values.web.service.port }}
    {{- end }}
    {{- if .Values.grafana.enabled }}
    Grafana     : http://localhost:{{ .Values.grafana.service.port }}  ({{ .Values.grafana.adminUser }} / <secret>)
    {{- end }}
{{- end }}

  Next steps
